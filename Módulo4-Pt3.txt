'Módulo 4 - Parte 3 de 4: detección de filas, importación y creación de hojas
'Entrega parcial para integración; no es el Módulo 4 completo ni funciona de forma independiente.
'En una copia, sustituir los quince procedimientos homónimos completos listados abajo.
'No pegar al final del original: quedarían nombres duplicados.
'Conservar las Partes 1 y 2, CLAVE_HOJA y los demás procedimientos originales.
'No ejecutar en producción hasta integrar las cuatro partes y probarlas en una copia.
'Procedimientos que sustituye esta parte:
'FilaTieneDatosEntrada, ObtenerUltimaFilaEntrada, ObtenerUltimaFilaParaRevision,
'CrearNuevaHojaDesdePlantilla, CopiarAnchosColumnas, CopiarAlturasFilas,
'CopiarFormasZonaSuperior, ObtenerHojaPlantilla, ObtenerHojaMenu, EsHojaTecnica,
'ContarHojasTrabajoCreadas, ExisteHoja, FilaMenuCompleta,
'AplicarFormulasEncabezadosDesdeMenu e ImportarDatosDesdeArchivoExcel.
'Entradas: A:G y U:V desde fila 12. H:T no se cuenta como entrada ni se borra/importa.
'La creación limpia solo A12:G500 y U12:V500: conserva las fórmulas copiadas de PLANTILLA.
'H debe existir previamente en PLANTILLA: =SI(B12=0, "", SUMA($B$2-B12))
'Esta parte no crea/restaura la fórmula: esa integración corresponde a la Parte 4.
'Importación: conserva la copia posicional de valores del rango continuo elegido por el usuario.
'El rango origen debe estar en el mismo orden que el bloque destino; no se transforma el archivo antiguo.
'Cada importación debe caber COMPLETA en A:G o en U:V, nunca cruzar H:T.
'Para llenar ambos bloques, realizar importaciones separadas; nunca seleccionar A:V completo.
'Si el origen usa el orden antiguo, seleccionar bloques compatibles por separado:
'A:B a A:B; C a G; D a D; E a C; F a U; G a E; H a F.
'V solo se importa si el origen dispone del dato Partida a reclasificar.
'Se conservan los límites originales: creación 12:500 e importación hasta la última fila de Excel.
'La protección sigue delegada en ProtegerHojaTrabajo (Parte 4).
'Código revisado estáticamente; no compilado ni ejecutado en Excel.
Function FilaTieneDatosEntrada(ByVal ws As Worksheet, ByVal fila As Long) As Boolean

    If Application.WorksheetFunction.CountA(ws.Range("A" & fila & ":G" & fila), ws.Range("U" & fila & ":V" & fila)) > 0 Then
        FilaTieneDatosEntrada = True
    Else
        FilaTieneDatosEntrada = False
    End If

End Function

Function ObtenerUltimaFilaEntrada(ByVal ws As Worksheet) As Long

    Dim ultimaCelda As Range
    Dim ultimaCeldaAdicional As Range

    Set ultimaCelda = ws.Range("A:G").Find(What:="*", _
                                         LookIn:=xlFormulas, _
                                         SearchOrder:=xlByRows, _
                                         SearchDirection:=xlPrevious)

    If ultimaCelda Is Nothing Then
        ObtenerUltimaFilaEntrada = 0
    Else
        ObtenerUltimaFilaEntrada = ultimaCelda.Row
    End If


    Set ultimaCeldaAdicional = ws.Range("U:V").Find(What:="*", _
                                                  LookIn:=xlFormulas, _
                                                  SearchOrder:=xlByRows, _
                                                  SearchDirection:=xlPrevious)

    If Not ultimaCeldaAdicional Is Nothing Then
        If ultimaCeldaAdicional.Row > ObtenerUltimaFilaEntrada Then
            ObtenerUltimaFilaEntrada = ultimaCeldaAdicional.Row
        End If
    End If
End Function

Function ObtenerUltimaFilaParaRevision(ByVal ws As Worksheet) As Long

    Dim ultimaFilaDatos As Long
    Dim ultimaFilaUsada As Long

    ultimaFilaDatos = ObtenerUltimaFilaEntrada(ws)

    With ws.UsedRange
        ultimaFilaUsada = .Row + .Rows.Count - 1
    End With

    If ultimaFilaDatos > ultimaFilaUsada Then
        ObtenerUltimaFilaParaRevision = ultimaFilaDatos
    Else
        ObtenerUltimaFilaParaRevision = ultimaFilaUsada
    End If

End Function

Sub CrearNuevaHojaDesdePlantilla()

    Dim wsMenu As Worksheet
    Dim wsPlantilla As Worksheet
    Dim wsNueva As Worksheet

    Dim visibilidadOriginal As XlSheetVisibility
    Dim numeroNuevaHoja As Long
    Dim filaMenu As Long
    Dim nombreNuevaHoja As String
    Dim detalleErrores As String

    Dim hojaCreada As Boolean
    Dim creacionFinalizada As Boolean

    Dim paso As String

    On Error GoTo ErrorHandler

    paso = "buscar hoja menú"
    Set wsMenu = ObtenerHojaMenu()

    If wsMenu Is Nothing Then
        MsgBox "No se encontró la hoja MENÚ.", _
               vbCritical, _
               "No se puede crear la hoja"
        Exit Sub
    End If

    On Error Resume Next

    paso = "Buscar hoja plantilla"
    Set wsPlantilla = ObtenerHojaPlantilla()

    If wsPlantilla Is Nothing Then
        MsgBox "No se encontró la hoja maestra llamada PLANTILLA" & vbCrLf & vbCrLf & _
               "Libro activo: " & ActiveWorkbook.Name & vbCrLf & vbCrLf & _
               "Debe existir una hoja llamada exactamente PLANTILLA", _
               vbCritical, _
               "No se pudo crear la hoja"
        Exit Sub
    End If

    numeroNuevaHoja = ContarHojasTrabajoCreadas() + 1
    filaMenu = numeroNuevaHoja + 9
    nombreNuevaHoja = "PLANTILLA " & Format(numeroNuevaHoja, "00")

    If Not FilaMenuCompleta(wsMenu, filaMenu, detalleErrores) Then

        MsgBox "No se puede crear la hoja " & nombreNuevaHoja & "." & vbCrLf & vbCrLf & _
               "La fila " & filaMenu & " de la hoja MENÚ no está completa en el rango C:L" & vbCrLf & vbCrLf & _
               "Debe corregir lo siguiente:" & vbCrLf & _
               detalleErrores, _
               vbCritical, _
               "MENÚ Incompleto"

        Exit Sub

    End If

    If ExisteHoja(nombreNuevaHoja) Then

        MsgBox "No se puede crear la hoja porque ya existe una hoja llamada: " & vbCrLf & vbCrLf & _
               nombreNuevaHoja & vbCrLf & vbCrLf & _
               "Revise esa hoja antes de continuar.", _
               vbCritical, _
               "Nombre de hoja existente"

        Exit Sub

    End If

    If MsgBox("Se creará la hoja:" & vbCrLf & vbCrLf & _
              nombreNuevaHoja & vbCrLf & vbCrLf & _
              "Usando la fila " & filaMenu & " de la hoja MENÚ." & vbCrLf & vbCrLf & _
              "¿Desea continuar?", _
              vbYesNo + vbQuestion, _
              "Crear nueva hoja") <> vbYes Then
        Exit Sub
    End If

    Application.ScreenUpdating = False
    Application.EnableEvents = False
    Application.DisplayAlerts = False

    visibilidadOriginal = wsPlantilla.Visible

    If wsPlantilla.Visible <> xlSheetVisible Then
        wsPlantilla.Visible = xlSheetVisible
    End If

    paso = "Copiar hoja plantilla"
    wsPlantilla.Copy After:=ActiveWorkbook.Worksheets(ActiveWorkbook.Worksheets.Count)

    hojaCreada = True
    Set wsNueva = ActiveSheet

    wsPlantilla.Visible = visibilidadOriginal

    paso = "Renombrar hoja nueva"
    wsNueva.Name = nombreNuevaHoja

    On Error Resume Next
    wsNueva.Unprotect Password:=CLAVE_HOJA
    On Error GoTo ErrorHandler

    paso = "Aplicar fórmulas encabezado"
    Call AplicarFormulasEncabezadosDesdeMenu(wsNueva, filaMenu)

    paso = "Eliminar contenido de A12:G500 y U12:V500"
    wsNueva.Range("A12:G500").ClearContents
    wsNueva.Range("U12:V500").ClearContents

    paso = "Configurar validaciones entrada en hoja"
    Call ConfigurarValidacionesEntradaEnHoja(wsNueva)

    paso = "Proteger hoja trabajo"
    Call ProtegerHojaTrabajo(wsNueva)

    creacionFinalizada = True

    MsgBox "Hoja creada correctamente." & vbCrLf & vbCrLf & _
           "Nombre: " & nombreNuevaHoja & vbCrLf & _
           "Fila usada en MENÚ: " & filaMenu, _
           vbInformation, _
           "Proceso finalizado"

SalidaSegura:

    On Error Resume Next

    If Not wsPlantilla Is Nothing Then
        wsPlantilla.Visible = visibilidadOriginal
    End If

    Application.DisplayAlerts = True
    Application.EnableEvents = True
    Application.ScreenUpdating = True

    On Error GoTo 0

    Exit Sub

ErrorHandler:

    Dim numeroError As Long
    Dim descripcionError As String

    numeroError = Err.Number
    descripcionError = Err.Description

    On Error Resume Next

    If hojaCreada = True And creacionFinalizada = False Then

        If Not wsNueva Is Nothing Then
            Application.DisplayAlerts = False
            wsNueva.Delete
            Application.DisplayAlerts = True
        End If

    End If

    If Not wsPlantilla Is Nothing Then
        wsPlantilla.Visible = visibilidadOriginal
    End If

    Application.DisplayAlerts = True
    Application.EnableEvents = True
    Application.ScreenUpdating = True

    On Error GoTo 0

    MsgBox "No se pudo crear la hoja nueva desde PLANTILLA." & vbCrLf & vbCrLf & _
           "Paso donde falló: " & paso & vbCrLf & vbCrLf & _
           "Número de error: " & Err.Number & vbCrLf & _
           "Descripción: " & Err.Description, _
           vbCritical, _
           "Error creando hoja"

End Sub

Sub CopiarAnchosColumnas(ByVal wsOrigen As Worksheet, _
                         ByVal wsDestino As Worksheet, _
                         ByVal columnaInicial As Long, _
                         ByVal columnaFinal As Long)

    Dim col As Long

    For col = columnaInicial To columnaFinal
        wsDestino.Columns(col).ColumnWidth = wsOrigen.Columns(col).ColumnWidth
    Next col

End Sub

Sub CopiarAlturasFilas(ByVal wsOrigen As Worksheet, _
                       ByVal wsDestino As Worksheet, _
                       ByVal filaInicial As Long, _
                       ByVal filaFinal As Long)

    Dim fila As Long

    For fila = filaInicial To filaFinal
        wsDestino.Rows(fila).RowHeight = wsOrigen.Rows(fila).RowHeight
    Next fila

End Sub

Sub CopiarFormasZonaSuperior(ByVal wsOrigen As Worksheet, _
                             ByVal wsDestino As Worksheet)

    Dim shp As Shape
    Dim nuevaForma As Shape
    Dim rangoZona As Range

    Set rangoZona = wsOrigen.Range("A1:U9")

    For Each shp In wsOrigen.Shapes

        If Not Intersect(shp.TopLeftCell, rangoZona) Is Nothing Then

            shp.Copy
            wsDestino.Paste

            Set nuevaForma = wsDestino.Shapes(wsDestino.Shapes.Count)

            nuevaForma.Top = shp.Top
            nuevaForma.Left = shp.Left
            nuevaForma.Width = shp.Width
            nuevaForma.Height = shp.Height

            'Conservar macro asignada si la forma tenía una macro'
            On Error Resume Next
            nuevaForma.OnAction = shp.OnAction
            On Error GoTo 0

        End If

    Next shp

End Sub

Function ObtenerHojaPlantilla() As Worksheet

    Dim wb As Workbook
    Dim ws As Worksheet

    Set ObtenerHojaPlantilla = Nothing
    Set wb = ActiveWorkbook

    For Each ws In wb.Worksheets

        If UCase(Trim(ws.Name)) = "PLANTILLA" Then
            Set ObtenerHojaPlantilla = ws
            Exit Function
        End If

    Next ws

End Function

Function ObtenerHojaMenu() As Worksheet

    Dim wb As Workbook
    Dim ws As Worksheet

    Set ObtenerHojaMenu = Nothing
    Set wb = ActiveWorkbook

    For Each ws In wb.Worksheets

        If UCase(Trim(ws.Name)) = "MENÚ" Or _
           UCase(Trim(ws.Name)) = "MENU" Then

            Set ObtenerHojaMenu = ws
            Exit Function
        End If

    Next ws

End Function

Function EsHojaTecnica(ByVal nombreHoja As String) As Boolean

    Select Case UCase(nombreHoja)

        Case "MENÚ", "MENU", "ALCON", "DATOS", "PLANTILLA"
            EsHojaTecnica = True

        Case Else
            EsHojaTecnica = False

    End Select
End Function

Function ContarHojasTrabajoCreadas() As Long

    Dim ws As Worksheet
    Dim contador As Long

    contador = 0

    For Each ws In ActiveWorkbook.Worksheets

        If Not EsHojaTecnica(ws.Name) Then
            contador = contador + 1
        End If

    Next ws

    ContarHojasTrabajoCreadas = contador

End Function

Function ExisteHoja(ByVal nombreHoja As String) As Boolean

    Dim ws As Worksheet

    ExisteHoja = False

    For Each ws In ActiveWorkbook.Worksheets

        If UCase(ws.Name) = UCase(nombreHoja) Then
            ExisteHoja = True
            Exit Function
        End If

    Next ws

End Function

Function FilaMenuCompleta(ByVal wsMenu As Worksheet, ByVal filaMenu As Long, ByRef detalleErrores As String) As Boolean

    Dim col As Long
    Dim direccionCelda As String
    Dim valorCelda As Variant

    FilaMenuCompleta = True
    detalleErrores = ""

    'Columna C hasta la L'
    For col = 3 To 12

        direccionCelda = wsMenu.Cells(filaMenu, col).Address(False, False)

        'Validar que la celda tenga fórmula'
        If wsMenu.Cells(filaMenu, col).HasFormula = False Then

            FilaMenuCompleta = False
            detalleErrores = detalleErrores & "- " & direccionCelda & " no tiene fórmula." & vbCrLf

        End If

    Next col

End Function

Sub AplicarFormulasEncabezadosDesdeMenu(ByVal wsNueva As Worksheet, ByVal filaMenu As Long)

    wsNueva.Range("B1").Formula = "='MENÚ'!I" & filaMenu

    wsNueva.Range("B4").Formula = "='MENÚ'!A" & filaMenu
    wsNueva.Range("B5").Formula = "='MENÚ'!B" & filaMenu
    wsNueva.Range("B6").Formula = "='MENÚ'!C" & filaMenu
    wsNueva.Range("B7").Formula = "='MENÚ'!H" & filaMenu
    wsNueva.Range("B8").Formula = "='MENÚ'!J" & filaMenu
    wsNueva.Range("B9").Formula = "='MENÚ'!D" & filaMenu

    wsNueva.Range("F7").Formula = "='MENÚ'!E" & filaMenu
    wsNueva.Range("G7").Formula = "='MENÚ'!F" & filaMenu

End Sub

Sub ImportarDatosDesdeArchivoExcel()

    Dim wsDestino As Worksheet
    Dim celdaDestino As Range
    Dim rangoEditable As Range
    Dim estabaProtegida As Boolean

    Dim rutaArchivo As Variant
    Dim wbOrigen As Workbook
    Dim rngOrigen As Range

    Dim filasOrigen As Long
    Dim columnasOrigen As Long

    Dim ultimaFilaPermitida As Long
    Dim ultimaColumnaPermitida As Long
    Dim filaFinalDestino As Long
    Dim columnaFinalDestino As Long

    On Error GoTo ErrorHandler

    Set wsDestino = ActiveSheet
    Set celdaDestino = ActiveCell
    Set rangoEditable = Union(wsDestino.Range("A12:G1048576"), wsDestino.Range("U12:V1048576"))

    If wsDestino.Name = "MENÚ" Then Exit Sub
    If wsDestino.Name = "ALCON" Then Exit Sub
    If wsDestino.Name = "DATOS" Then Exit Sub
    If wsDestino.Name = "PLANTILLA" Then Exit Sub

    'Validar que el usuario esté ubicado dentro de la zona editable'
    If Intersect(celdaDestino, rangoEditable) Is Nothing Then

        MsgBox "Antes de importar, seleccione la celda inicial dentro de la zona editable A12:G1048576 o U12:V1048576.", _
               vbExclamation, _
               "Importar datos"

        Exit Sub

    End If

    'Cada importación debe quedar dentro de un solo bloque editable, sin tocar H:T'
    If celdaDestino.Column <= 7 Then
        Set rangoEditable = wsDestino.Range("A12:G1048576")
    Else
        Set rangoEditable = wsDestino.Range("U12:V1048576")
    End If

    estabaProtegida = wsDestino.ProtectContents

    If estabaProtegida Then
        wsDestino.Unprotect Password:=CLAVE_HOJA
    End If

    'Seleccionar archivo origen'
    rutaArchivo = Application.GetOpenFilename( _
                    FileFilter:="Archivos de Excel (*.xlsx; *.xlsm; *.xls), *.xlsx; *.xlsm; *.xls", _
                    Title:="Seleccione el archivo Excel origen")

    If rutaArchivo = False Then GoTo SalidaSegura

    'Evitar importar el mismo archivo actual'
    If CStr(rutaArchivo) = ThisWorkbook.FullName Then

        MsgBox "No seleccione el mismo archivo de la plantilla como archivo origen.", _
               vbExclamation, _
               "Archivo no válido"

        GoTo SalidaSegura

    End If

    Application.ScreenUpdating = True
    Application.EnableEvents = False
    Application.DisplayAlerts = False

    'Abrir archivo origen solo lectura'
    Set wbOrigen = Workbooks.Open(Filename:=CStr(rutaArchivo), ReadOnly:=True)

    MsgBox "Ahora seleccione con el mouse el rango de datos que desea importar en el archivo origen.", _
           vbInformation, _
           "Seleccionar rango origen"

    'Permitir que el usuario seleccione el rango origen'
    On Error Resume Next
    Set rngOrigen = Application.InputBox( _
                        Prompt:="Seleccione el rango origen a importar.", _
                        Title:="Rango origen", _
                        Type:=8)
    On Error GoTo ErrorHandler

    If rngOrigen Is Nothing Then GoTo SalidaSegura

    'No permitir rangos separados'
    If rngOrigen.Areas.Count > 1 Then

        MsgBox "Seleccione un solo rango continuo. No se permiten rangos separados.", _
               vbExclamation, _
               "Rango no válido"

        GoTo SalidaSegura

    End If

    filasOrigen = rngOrigen.Rows.Count
    columnasOrigen = rngOrigen.Columns.Count

    'Validar límites de destino'
    ultimaFilaPermitida = rangoEditable.Row + rangoEditable.Rows.Count - 1
    ultimaColumnaPermitida = rangoEditable.Column + rangoEditable.Columns.Count - 1

    filaFinalDestino = celdaDestino.Row + filasOrigen - 1
    columnaFinalDestino = celdaDestino.Column + columnasOrigen - 1

    If filaFinalDestino > ultimaFilaPermitida Or columnaFinalDestino > ultimaColumnaPermitida Then

        MsgBox "El rango seleccionado no cabe dentro de la zona editable A12:G1048576 o U12:V1048576." & vbCrLf & vbCrLf & _
               "Filas seleccionadas: " & filasOrigen & vbCrLf & _
               "Columnas seleccionadas: " & columnasOrigen & vbCrLf & _
               "Celda inicial destino: " & celdaDestino.Address(False, False) & vbCrLf & vbCrLf & _
               "Seleccione menos datos o cambie la celda inicial de destino.", _
               vbCritical, _
               "Importación fuera de rango"

        GoTo SalidaSegura

    End If

    'Volver a la hoja destino'
    wsDestino.Activate

    'Pegar únicamente valores en el mismo orden del rango seleccionado'
    'Para archivos antiguos, seleccionar por separado los datos que coincidan con el destino'
    wsDestino.Range(celdaDestino.Address).Resize(filasOrigen, columnasOrigen).Value = rngOrigen.Value

    MsgBox "Datos importados correctamente como valores." & vbCrLf & vbCrLf & _
           "Filas importadas: " & filasOrigen & vbCrLf & _
           "Columnas importadas: " & columnasOrigen, _
           vbInformation, _
           "Importación exitosa"

SalidaSegura:

    On Error Resume Next

    If Not wbOrigen Is Nothing Then
        wbOrigen.Close SaveChanges:=False
    End If

    If Not wsDestino Is Nothing Then
        Call ProtegerHojaTrabajo(wsDestino)
    End If

    If estabaProtegida Then
        Call ProtegerHojaTrabajo(wsDestino)
    End If

    Application.DisplayAlerts = True
    Application.EnableEvents = True
    Application.ScreenUpdating = True

    On Error GoTo 0

    Exit Sub

ErrorHandler:

    MsgBox "No se pudieron importar los datos." & vbCrLf & vbCrLf & _
           "Número de error: " & Err.Number & vbCrLf & _
           "Descripción: " & Err.Description, _
           vbCritical, _
           "Error al importar"

    Resume SalidaSegura

End Sub
