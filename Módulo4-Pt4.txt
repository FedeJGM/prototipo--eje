'Módulo 4 - Parte 4 de 4: protección y fórmula de H
'Entrega SOLO de la Parte 4; no es el módulo integrado ni funciona de forma independiente.
'En una copia, sustituir completos ProtegerTodasLasHojasTrabajo y ProtegerHojaTrabajo.
'No pegar al final: duplicaría procedimientos. Conservar CLAVE_HOJA y las Partes 1, 2 y 3.
'Se conserva el límite original de protección: filas 12 a 500, NO toda la columna.
'A12:G500 y U12:V500 editables; H12:H500 calculada y bloqueada.
'Los bloqueos y las fórmulas fuera de esas zonas no se modifican.
'Al proteger se restablece H en 12:500, incluso si allí faltaba la fórmula.
'H12: =SI(B12=0, "", SUMA($B$2-B12)); B2 Fecha corte y B12 Fecha origen.
'FormulaLocal usa SI/SUMA de Excel en español y adapta coma/punto y coma al separador regional.
'Excel puede normalizar los espacios/separadores al mostrar la misma fórmula.
'Requiere Excel con nombres de funciones en español; no usa una fórmula inglesa sustitutiva.
'Pendiente de integración: corregir las salidas de validación/importación que omiten reproteger.
'Esta parte intenta proteger ante un error propio, pero no garantiza las salidas de otros procedimientos.
'No ejecutar en producción: revisar límites de filas e integrar/probar primero en una copia.
'Comprobación estática únicamente; no compilado ni ejecutado en Excel.

'Validación para proteger todas las hojas del libro Excel'
Sub ProtegerTodasLasHojasTrabajo()

    Dim ws As Worksheet
    Dim respuesta As VbMsgBoxResult
    Dim pantallaAnterior As Boolean
    Dim eventosAnteriores As Boolean
    Dim alertasAnteriores As Boolean

    respuesta = MsgBox("Esta macro protegerá todas las hojas permitidas: " & vbCrLf & vbCrLf & _
    "Configuraciones aplicadas:" & vbCrLf & _
    "A12:G500 y U12:V500 editables" & vbCrLf & _
    "H12:H500 con fórmula y bloqueada" & vbCrLf & _
    "Los demás bloqueos y fórmulas se conservan" & vbCrLf & _
    "Hoja protegida" & vbCrLf & vbCrLf & _
    "¿Desea continuar?", _
    vbYesNo + vbExclamation, _
    "Proteger hojas")

    If respuesta <> vbYes Then Exit Sub

    pantallaAnterior = Application.ScreenUpdating
    eventosAnteriores = Application.EnableEvents
    alertasAnteriores = Application.DisplayAlerts

    On Error GoTo ErrorHandler

    Application.ScreenUpdating = False
    Application.EnableEvents = False
    Application.DisplayAlerts = False

    For Each ws In ThisWorkbook.Worksheets

        If Not EsHojaTecnica(ws.Name) Then
            Call ProtegerHojaTrabajo(ws)
        End If

    Next ws

    MsgBox "Protección aplicada correctamente en todas las hojas.", _
        vbInformation, _
        "Proceso finalizado"

SalidaSegura:

    Application.ScreenUpdating = pantallaAnterior
    Application.EnableEvents = eventosAnteriores
    Application.DisplayAlerts = alertasAnteriores
    Exit Sub

ErrorHandler:

    MsgBox "No se completó la protección de todas las hojas." & vbCrLf & vbCrLf & _
        "Detalle: " & Err.Description, _
        vbCritical, _
        "Error protegiendo las hojas"

    Resume SalidaSegura

End Sub

'Validación Llamar para proteger hoja actual'
Sub ProtegerHojaTrabajo(ByVal ws As Worksheet)

    Dim fila As Long
    Dim separador As String
    Dim numeroError As Long
    Dim descripcionError As String
    Dim errorProteccion As String

    On Error GoTo ErrorHandler

    'No modificar hojas auxiliares'
    If EsHojaTecnica(ws.Name) Then Exit Sub

    'Desproteger primero; si falla, no continuar modificando celdas'
    ws.Unprotect Password:=CLAVE_HOJA

    'Conservar los bloqueos existentes fuera de las zonas aprobadas'
    ws.Range("A12:G500").Locked = False
    ws.Range("U12:V500").Locked = False
    ws.Range("H12:H500").Locked = True

    'Restablecer solamente la fórmula aprobada de H'
    separador = Application.International(xlListSeparator)

    For fila = 12 To 500
        ws.Cells(fila, "H").FormulaLocal = Replace( _
            "=SI(B" & fila & "=0, """", SUMA($B$2-B" & fila & "))", ",", separador)
    Next fila

    'Proteger hoja'
    ws.Protect Password:=CLAVE_HOJA, _
        UserInterfaceOnly:=True, _
        AllowFiltering:=True

    Exit Sub

ErrorHandler:

    numeroError = Err.Number
    descripcionError = Err.Description

    'Intentar recuperar la protección aunque falle la actualización de H'
    On Error Resume Next
    Err.Clear

    ws.Protect Password:=CLAVE_HOJA, _
        UserInterfaceOnly:=True, _
        AllowFiltering:=True

    If Err.Number <> 0 Then
        errorProteccion = vbCrLf & "También falló el intento de proteger: " & Err.Description
    End If

    On Error GoTo 0

    'Propagar el error: el llamador no debe informar éxito si no se completó'
    Err.Raise numeroError, "ProtegerHojaTrabajo", _
        "Hoja: " & ws.Name & vbCrLf & descripcionError & errorProteccion

End Sub
