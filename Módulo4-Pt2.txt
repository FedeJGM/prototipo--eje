'La lista de D contiene las ocho Redes del catálogo original y de las reglas de negocio.

'Catálogo de NIT
Function ObtenerNitRed(ByVal nombreRed As Variant) As String

    Dim red As String
    red = LimpiarTextoValidacion(nombreRed)
    Select Case red

        Case "AMEXINTERNACIONAL"
            ObtenerNitRed = "4001434464"

        Case "REDEBAN"
            ObtenerNitRed = "830070527"

        Case "MASTERCARDINTERNACIONAL"
            ObtenerNitRed = "830002631"

        Case "MASTERCARDCOLOMBIAADMINISTRADORA"
            ObtenerNitRed = "901010458"

        Case "CREDIBANCOAMEX"
            ObtenerNitRed = "860032909"

        Case "CREDIBANCOVISA"
            ObtenerNitRed = "860032909"

        Case "VISAINTERNACIONAL"
            ObtenerNitRed = "900878539"

        Case "VISANACIONAL"
            ObtenerNitRed = "900550459"

        Case Else
            ObtenerNitRed = ""

    End Select
End Function

End Function

Function EsNombreRedCatalogo(ByVal valor As Variant) As Boolean

    If ObtenerNitRed(valor) <> "" Then
        EsNombreRedCatalogo = True
    Else
        EsNombreRedCatalogo = False
    End If

End Function

Function EsNitRedCatalogo(ByVal valor As Variant) As Boolean

    Dim nit As String
    EsNitRedCatalogo = False

    If IsError(valor) Then Exit Function

    nit = Trim(CStr(valor))
    nit = Replace(nit, " ", "")
    nit = Replace(nit, Chr(160), "")

    Select Case nit

        Case "4001434464"
            EsNitRedCatalogo = True

        Case "830070527"
            EsNitRedCatalogo = True

        Case "830002631"
            EsNitRedCatalogo = True

        Case "901010458"
            EsNitRedCatalogo = True

        Case "860032909"
            EsNitRedCatalogo = True

        Case "900878539"
            EsNitRedCatalogo = True

        Case "900550459"
            EsNitRedCatalogo = True

    End Select
End Function

Sub AplicarListaRedesEnCelda(ByVal celda As Range)

    On Error Resume Next
    celda.Validation.Delete
    On Error GoTo 0

    With celda.Validation
        .Add Type:=xlValidateList, _
             AlertStyle:=xlValidAlertStop, _
             Operator:=xlBetween, _
             Formula1:="Amex Internacional,Redeban,Mastercard Internacional,Mastercard Colombia Administradora,Credibanco Amex,Credibanco Visa,Visa Internacional,Visa Nacional"

        .IgnoreBlank = True
        .InCellDropdown = True

        .InputTitle = "Nombre de la Red"
        .InputMessage = "Selecciona una Red de la lista"

        .ErrorTitle = "Red Inválida"
        .ErrorMessage = "Debe seleccionar una Red válida de la lista"

        .ShowInput = True
        .ShowError = True
    End With

End Sub

Sub QuitarListaRedesEnCelda(ByVal celda As Range)

    On Error Resume Next
    celda.Validation.Delete
    On Error GoTo 0

End Sub

Sub ActualizarListaRedSegunTipoCliente(ByVal ws As Worksheet, ByVal Target As Range)

    Dim rangoCambio As Range
    Dim celda As Range
    Dim fila As Long
    Dim tipoCliente As String

    Set rangoCambio = Intersect(Target, Union(ws.Range("G12:G" & ws.Rows.Count), ws.Range("D12:D" & ws.Rows.Count)))

    If rangoCambio Is Nothing Then Exit Sub

    For Each celda In rangoCambio.Cells

        fila = celda.Row
        tipoCliente = LimpiarTextoValidacion(ws.Cells(fila, "G").Value)

        If tipoCliente = "RED" Then

            Call AplicarListaRedesEnCelda(ws.Cells(fila, "D"))

            If ObtenerNitRed(ws.Cells(fila, "D").Value) <> "" Then
                ws.Cells(fila, "C").Value = ObtenerNitRed(ws.Cells(fila, "D").Value)
                ws.Cells(fila, "U").Value = ObtenerOficinaRed(ws.Cells(fila, "D").Value)
            End If

        Else

            Call QuitarListaRedesEnCelda(ws.Cells(fila, "D"))

        End If

    Next celda
End Sub
