'Conservar CLAVE_HOJA y los demás procedimientos originales.
'No ejecutar en producción hasta integrar las partes 2, 3 y 4.
'Pendientes: Redes, detección de filas/importación y protección/fórmula H.
'La fórmula de H no se escribe ni se borra aquí.
'Se elimina la validación antigua de H, no su contenido ni fórmula.
'V: lista Sí/No con IgnoreBlank=True, sin imponer obligatoriedad nueva.

Sub ValidarTodo_ReglasNegocio()

    Dim ws As Worksheet
    Dim ultimaFila As Long
    Dim fila As Long
    Dim errores As Long
    Dim valor As Variant
    Dim estabaProtegida As Boolean

    Set ws = ActiveSheet

    If estabaProtegida Then
        ws.Unprotect Password:=CLAVE_HOJA

        If ws.ProtectContents Then

            MsgBox "No se pudo desproteger la hoja" & ws.Name & vbCrLf & vbCrLf & _
                   "La validación no se pudo continuar porque la hoja sigue protegida", _
                   vbCritical, _
                   "Hoja Protegida"

            GoTo SalidaSegura
        End If

    End If

    'No validar hojas auxiliares'
    If ws.Name = "ALCON" Then
        MsgBox "Esta hoja no debe validarse.", vbExclamation, "Validación"
        Exit Sub
    End If

    If ws.Name = "MENÚ" Or ws.Name = "MENU" Or ws.Name = "Menú" Or ws.Name = "Menu" Or ws.Name = "menu" Then
        MsgBox "Esta hoja no debe validarse.", vbExclamation, "Validación"
        Exit Sub
    End If

    If ws.Name = "PLANTILLA" Then
        MsgBox "Esta no debe validarse.", vbExclamation, "Validación"
        Exit Sub
    End If

    Application.ScreenUpdating = False
    Application.EnableEvents = False

    ultimaFila = ObtenerUltimaFilaParaRevision(ws)

    If ultimaFila < 12 Then
        MsgBox "No hay datos para validar desde la fila 12.", vbInformation, "Validación"
        GoTo SalidaSegura
    End If

    Call LimpiarColoresZonaEditable(ws)

    For fila = 12 To 500

        'Validar Columna A: Código Oficina'
        If FilaTieneDatosEntrada(ws, fila) Then

            If Not EsCodigoOficinaValido(ws.Cells(fila, "A").Value) Then
                ws.Cells(fila, "A").Interior.Color = RGB(255, 0, 0)
                errores = errores + 1
            End If
        End If

        'Validar columna B - Fecha origen partida'
        If FilaTieneDatosEntrada(ws, fila) Then

            If Not EsFechaOrigenPartidaValida(ws.Cells(fila, "B")) Then
                ws.Cells(fila, "B").Interior.Color = RGB(255, 0, 0)
                errores = errores + 1
            End If
        End If

        'Validar columna G - Tipo de cliente'
        If FilaTieneDatosEntrada(ws, fila) Then

            If Not EsTipoClienteValido(ws.Cells(fila, "G").Value) Then
                ws.Cells(fila, "G").Interior.Color = RGB(255, 0, 0)
                errores = errores + 1
            End If

        End If

        'Regla dependiente cuando tipo Cliente = Red'
        If FilaTieneDatosEntrada(ws, fila) Then

            If LimpiarTextoValidacion(ws.Cells(fila, "G").Value) = "RED" Then

                'Si G = Red, D debe ser una red válida en el catálogo'
                If ObtenerNitRed(ws.Cells(fila, "D").Value) <> "" Then

                    'Rellenar automáticamente NIT y Oficina de Reclasificación'
                    ws.Cells(fila, "C").Value = ObtenerNitRed(ws.Cells(fila, "D").Value)
                    ws.Cells(fila, "U").Value = ObtenerOficinaRed(ws.Cells(fila, "D").Value)

                Else

                    'Si G es Red, pero D no corresponde a una Red válida'
                    ws.Cells(fila, "D").Interior.Color = RGB(255, 0, 0)
                    ws.Cells(fila, "C").ClearContents
                    ws.Cells(fila, "U").ClearContents
                    errores = errores + 1

                End If

            Else

                'Si G no es RED, D se valida como Nombre del cliente obligatorio'
                If Not EsNombreClienteValido(ws.Cells(fila, "D").Value) Then
                    ws.Cells(fila, "D").Interior.Color = RGB(255, 0, 0)
                    errores = errores + 1
                End If

            End If

        End If

        'Validar columna C - NIT'
        If FilaTieneDatosEntrada(ws, fila) Then

            If Not EsNitValido(ws.Cells(fila, "C").Value) Then
                ws.Cells(fila, "C").Interior.Color = RGB(255, 0, 0)
                errores = errores + 1

            End If

        End If

        'Validar inconsistencias: Tipo Cliente = Clientes pero datos pertenecen a Red'
        If FilaTieneDatosEntrada(ws, fila) Then

            If LimpiarTextoValidacion(ws.Cells(fila, "G").Value) = "CLIENTES" Then

                If EsNombreRedCatalogo(ws.Cells(fila, "D").Value) Then

                    ws.Cells(fila, "G").Interior.Color = RGB(255, 0, 0)
                    ws.Cells(fila, "D").Interior.Color = RGB(255, 0, 0)

                    errores = errores + 1

                End If

                If EsNitRedCatalogo(ws.Cells(fila, "C").Value) Then

                    ws.Cells(fila, "G").Interior.Color = RGB(255, 0, 0)
                    ws.Cells(fila, "C").Interior.Color = RGB(255, 0, 0)

                    errores = errores + 1

                End If

            End If

        End If

        'Validar columna U - Oficina reclasificación'
        If FilaTieneDatosEntrada(ws, fila) Then

            If Not EsOficinaReclasificacionValida(ws.Cells(fila, "U").Value) Then
                ws.Cells(fila, "U").Interior.Color = RGB(255, 0, 0)
                errores = errores + 1
            End If

        End If

        'Validar columna E - Descripción de la partida'
        If FilaTieneDatosEntrada(ws, fila) Then

            If Not EsDescripcionPartidaValida(ws.Cells(fila, "E").Value) Then
                ws.Cells(fila, "E").Interior.Color = RGB(255, 0, 0)
                errores = errores + 1
            End If

        End If

        'Validar columna F - Valor Partida'
        If FilaTieneDatosEntrada(ws, fila) Then

            If Not EsValorPartidaValido(ws.Cells(fila, "F")) Then
                ws.Cells(fila, "F").Interior.Color = RGB(255, 0, 0)
                errores = errores + 1
            End If

        End If

        'Validar columna V - Partida a reclasificar (si tiene valor)'
        If FilaTieneDatosEntrada(ws, fila) Then
            If Not EsPartidaReclasificarValida(ws.Cells(fila, "V").Value) Then
                ws.Cells(fila, "V").Interior.Color = RGB(255, 0, 0)
                errores = errores + 1
            End If
        End If

    Next fila

SalidaSegura:

    If estabaProtegida Then
        Call ProtegerHojaTrabajo(ws)
    End If

    Application.EnableEvents = True
    Application.ScreenUpdating = True

    If errores > 0 Then
        MsgBox "Validación finalizada. Se encontraron " & errores & " error(es). Las celdas inválidas fueron marcadas en rojo.", _
        vbExclamation, _
        "Validación reglas de negocio"

    Else
        MsgBox "Validación Exitosa. No se encontraron errores en las columnas.", _
        vbInformation, _
        "Validación reglas de negocio"
    End If
End Sub

Sub LimpiarColoresZonaEditable(ByVal ws As Worksheet)

    Dim celda As Range
    Dim rangoLimpiar As Range
    Dim direccionError As String

    On Error GoTo ErrorLimpiar

    Set rangoLimpiar = Union(ws.Range("A12:G500"), ws.Range("U12:V500"))

    For Each celda In rangoLimpiar.Cells

        direccionError = celda.Address(False, False)

        If celda.Interior.Color = RGB(255, 0, 0) Then

            If celda.MergeCells Then
                celda.MergeArea.Interior.ColorIndex = xlColorIndexNone

            Else
                celda.Interior.ColorIndex = xlColorIndexNone

            End If

        End If

    Next celda

    Exit Sub

ErrorLimpiar:

    MsgBox "Error Limpiando colores en la hoja" & ws.Name & vbCrLf & vbCrLf & _
           "Celda aproximada dónde falló: " & direccionError & vbCrLf & _
           "Detalle: " & Err.Description, _
           vbCritical, _
           "Error limpiando colores"

End Sub

Function ExistenErroresMarcadosEnRojo(ByVal ws As Worksheet) As Boolean

    Dim ultimaFila As Long
    Dim rangoRevision As Range
    Dim celda As Range

    ExistenErroresMarcadosEnRojo = False

    ultimaFila = ObtenerUltimaFilaParaRevision(ws)

    If ultimaFila < 12 Then Exit Function

    Set rangoRevision = Union(ws.Range("A12:G" & ultimaFila), ws.Range("U12:V" & ultimaFila))

    For Each celda In rangoRevision.Cells

        If celda.Interior.Color = RGB(255, 0, 0) Then
            ExistenErroresMarcadosEnRojo = True
            Exit Function
        End If

    Next celda

End Function

Function EsCodigoOficinaValido(ByVal valor As Variant) As Boolean

    EsCodigoOficinaValido = False
    If IsError(valor) Then Exit Function

    If Trim(CStr(valor)) = "" Then Exit Function

    If Not IsNumeric(valor) Then Exit Function

    If CDbl(valor) <> CLng(valor) Then Exit Function

    If CLng(valor) < 1 Or CLng(valor) > 999 Then Exit Function

    EsCodigoOficinaValido = True

End Function

Function EsFechaOrigenPartidaValida(ByVal celda As Range) As Boolean

    Dim fechaEvaluada As Date

    EsFechaOrigenPartidaValida = False

    If IsError(celda.Value) Then Exit Function

    If Trim(CStr(celda.Value)) = "" Then Exit Function

    If Not IsDate(celda.Value) Then Exit Function

    fechaEvaluada = DateValue(celda.Value)

    If fechaEvaluada < Date Then
        EsFechaOrigenPartidaValida = True
    End If

End Function

Function EsTipoClienteValido(ByVal valor As Variant) As Boolean

    Dim texto As String

    EsTipoClienteValido = False

    texto = LimpiarTextoValidacion(valor)

    If texto = "CLIENTES" Or texto = "RED" Or texto = "Clientes" Or texto = "Red" Then
        EsTipoClienteValido = True
    End If

End Function

Function EsNitValido(ByVal valor As Variant) As Boolean

    Dim texto As String
    Dim i As Long

    EsNitValido = False

    If IsError(valor) Then Exit Function

    texto = Trim(CStr(valor))

    'Si NIT no puede estar vacío si la fila tiene datos'
    If texto = "" Then Exit Function

    'Quitar espacios internos por seguridad'
    texto = Replace(texto, " ", "")

    'No permitir decimales, letras, guiones ni caracteres especiales'
    For i = 1 To Len(texto)
        If Mid(texto, i, 1) < "0" Or Mid(texto, i, 1) > "9" Then
            Exit Function
        End If
    Next i

    'Longitud razonable del NIT'
    If Len(texto) < 1 Or Len(texto) > 100 Then Exit Function

    EsNitValido = True

End Function

Function EsNombreClienteValido(ByVal valor As Variant) As Boolean

    Dim texto As String

    EsNombreClienteValido = False

    If IsError(valor) Then Exit Function

    texto = Trim(CStr(valor))

    If texto = "" Then Exit Function

    EsNombreClienteValido = True

End Function

Function EsOficinaReclasificacionValida(ByVal valor As Variant) As Boolean

    Dim texto As String

    EsOficinaReclasificacionValida = False

    If IsError(valor) Then Exit Function

    texto = Trim(CStr(valor))

    If texto = "" Then Exit Function

    texto = Replace(texto, " ", "")
    texto = Replace(texto, Chr(160), "")

    If texto = "977" Or texto = "979" Or texto = "995" Then
        EsOficinaReclasificacionValida = True
    End If

End Function

Function EsDescripcionPartidaValida(ByVal valor As Variant) As Boolean

    Dim texto As String

    EsDescripcionPartidaValida = False

    If IsError(valor) Then Exit Function

    texto = Trim(CStr(valor))

    If texto = "" Then Exit Function

    EsDescripcionPartidaValida = True

End Function

Function EsValorPartidaValido(ByVal celda As Range) As Boolean

    Dim textoOriginal As String
    Dim texto As String
    Dim i As Long
    Dim caracter As String
    Dim cantidadPuntos As Long
    Dim cantidadDigitos As Long
    Dim valorNumerico As Double

    EsValorPartidaValido = False

    If celda Is Nothing Then Exit Function

    If IsError(celda.Value) Then Exit Function

    texto = Trim(CStr(celda.Formula))
    texto = Replace(texto, Chr(160), "")
    texto = Replace(texto, " ", "")

    If texto = "" Then Exit Function

    'No permitir coma en la celda'
    If InStr(1, texto, ",", vbTextCompare) > 0 Then Exit Function

    'Permitir signo negativo al principio'
    If Left(texto, 1) = "-" Then
        texto = Mid(texto, 2)
    End If

    If texto = "" Then Exit Function

    For i = 1 To Len(texto)

        caracter = Mid(texto, i, 1)

        If caracter >= "0" And caracter <= "9" Then

            cantidadDigitos = cantidadDigitos + 1

        ElseIf caracter = "." Then

            cantidadPuntos = cantidadPuntos + 1

            If cantidadPuntos > 1 Then
                Exit Function
            End If
        Else

            Exit Function

        End If
    Next i

    'Debe tener al menos un número'
    If cantidadDigitos = 0 Then Exit Function

    'No permitir punto sólo al principio y final'
    If Left(texto, 1) = "." Then Exit Function

    If Right(texto, 1) = "." Then Exit Function

    valorNumerico = Val(texto)

    If valorNumerico = 0 Then Exit Function

    EsValorPartidaValido = True

End Function

Function LimpiarTextoValidacion(ByVal valor As Variant) As String

    Dim texto As String

    If IsError(valor) Then
        LimpiarTextoValidacion = ""
        Exit Function
    End If

    texto = CStr(valor)

    'Limpiar espacios normale, espacios invisibles y saltos de linea'
    texto = Replace(texto, Chr(160), "")
    texto = Replace(texto, vbCr, "")
    texto = Replace(texto, vbLf, "")

    'Quitar espacios al inicio y al final'
    texto = Trim(texto)

    Do While InStr(texto, " ") > 0
        texto = Replace(texto, " ", "")
    Loop

    LimpiarTextoValidacion = UCase(texto)

End Function

Sub LimpiarValidacionesEntradaHojaActiva()

    Dim ws As Worksheet

    Set ws = ActiveSheet

    If ws.Name = "ALCON" Then Exit Sub
    If ws.Name = "MENÚ" Then Exit Sub
    If ws.Name = "DATOS" Then Exit Sub
    If ws.Name = "PLANTILLA" Then Exit Sub

    'Limpiar validaciones de datos solo en la zona editable principal'
    ws.Range("A12:H" & ws.Rows.Count).Validation.Delete
    ws.Range("U12:V" & ws.Rows.Count).Validation.Delete

    MsgBox "Validaciones anteriores limpiadas en A:H y U:V desde la fila 12", _
           vbInformation, _
           "Limpieza de validaciones"

End Sub

Sub ConfigurarValidacionesEntradaHojaActiva()

    Dim ws As Worksheet
    Dim estabaProtegida As Boolean
    Dim errores As Long

    On Error GoTo ErrorGeneral

    If TypeName(ActiveSheet) <> "Worksheet" Then
        MsgBox "La hoja activa no es una hoja normal de Excel", vbCritical, "Error"
        Exit Sub
    End If

    Set ws = ActiveSheet

    If ws.Name = "ALCON" Then Exit Sub
    If ws.Name = "MENÚ" Then Exit Sub
    If ws.Name = "DATOS" Then Exit Sub
    If ws.Name = "PLANTILLA" Then Exit Sub

    Application.ScreenUpdating = False
    Application.EnableEvents = False

    estabaProtegida = ws.ProtectContents

    If estabaProtegida Then
        ws.Unprotect Password:=CLAVE_HOJA
    End If

    ws.Range("C12:C500").Validation.Delete
    ws.Range("H12:H500").Validation.Delete

    errores = 0

    If Not ConfigurarCodigoOficina(ws) Then
        errores = errores + 1
    End If

    If Not ConfigurarFechaOrigenPartida(ws) Then
        errores = errores + 1
    End If

    If Not ConfigurarTipoCliente(ws) Then
        errores = errores + 1
    End If

    If Not ConfigurarOficinaReclasificacion(ws) Then
        errores = errores + 1
    End If

    If Not ConfigurarValorPartida(ws) Then
        errores = errores + 1
    End If

    If Not ConfigurarPartidaReclasificar(ws) Then
        errores = errores + 1
    End If

SalidaSegura:

    If estabaProtegida Then
        Call ProtegerHojaTrabajo(ws)
    End If

    Application.EnableEvents = True
    Application.ScreenUpdating = True

    If errores = 0 Then
        MsgBox "Validaciones de entrada configuradas correctamente", vbInformation, "Validaciones"
    Else
        MsgBox "Las validaciones NO quedaron configuradas correctamente" & vbCrLf & vbCrLf & _
               "Columnas con errores en: " & errores & vbCrLf & vbCrLf & _
               "Revise los mensajes anteriores para identificar la columna afectada", _
               vbCritical, _
               "Validaciones con error"

    End If

    Exit Sub

ErrorGeneral:

    Application.EnableEvents = True
    Application.ScreenUpdating = True

    MsgBox "Error general configurando validaciones" & vbCrLf & vbCrLf & _
           "Detalle: " & Err.Description, _
           vbCritical, _
           "Error"

End Sub

Function ConfigurarCodigoOficina(ByVal ws As Worksheet) As Boolean

    Dim rangoCodigo As Range

    On Error GoTo ErrorConfigurarCodigo

    ConfigurarCodigoOficina = False

    Set rangoCodigo = ws.Cells(12, "A").Resize(500, 1)

    'Eliminar validaciones anteriores solo en Columna A'

    On Error Resume Next
    rangoCodigo.Validation.Delete
    On Error GoTo ErrorConfigurarCodigo

    With rangoCodigo.Validation
        .Add Type:=xlValidateWholeNumber, _
             AlertStyle:=xlValidAlertStop, _
             Operator:=xlBetween, _
             Formula1:="1", _
             Formula2:="999"

        .IgnoreBlank = True
        .InCellDropdown = False

        .InputTitle = "Código oficina"
        .InputMessage = "Ingrese un número entre 1 y 999"

        .ErrorTitle = "Código oficina inválido"
        .ErrorMessage = "Solo se permiten números naturales entre 1 y 999"

        .ShowInput = True
        .ShowError = True
    End With

    ConfigurarCodigoOficina = True
    Exit Function

ErrorConfigurarCodigo:
    MsgBox "Error configurando la validación de Código oficina en la hoja: " & ws.Name & vbCrLf & vbCrLf & _
           "Detalle: " & Err.Description, _
           vbCritical, _
           "Error Columna A"

End Function

Function ConfigurarFechaOrigenPartida(ByVal ws As Worksheet) As Boolean

    Dim rangoFecha As Range
    Dim fechaLimite As Long

    On Error GoTo ErrorConfigurarFecha

    ConfigurarFechaOrigenPartida = False

    Set rangoFecha = ws.Cells(12, "B").Resize(500, 1)

    'Fecha limite como numero serial en excel'
    fechaLimite = CLng(Date)

    On Error Resume Next
    rangoFecha.Validation.Delete
    On Error GoTo ErrorConfigurarFecha

    rangoFecha.NumberFormat = "dd/mm/yyyy"

    With rangoFecha.Validation
        .Add Type:=xlValidateDate, _
             AlertStyle:=xlValidAlertStop, _
             Operator:=xlLess, _
             Formula1:=CStr(fechaLimite)

        .IgnoreBlank = True
        .InCellDropdown = False
        .InputTitle = "Fecha origen partida"
        .InputMessage = "Ingrese una fecha anterior al dia actual en formato Día/Mes/Año"
        .ErrorTitle = "Fecha inválida"
        .ErrorMessage = "Fecha origen partida debe ser una fecha válida anterior al día actual. No se permite la fecha actual ni"
        .ShowInput = True
        .ShowError = True
    End With

    ConfigurarFechaOrigenPartida = True
    Exit Function

ErrorConfigurarFecha:

    MsgBox "Error configurando la validación Fecha origen partida en la hoja: " & ws.Name & vbCrLf & vbCrLf & _
           "Detalle: " & Err.Description, _
           vbCritical, _
           "Error Columna B"

End Function

Function ConfigurarValorPartida(ByVal ws As Worksheet) As Boolean

    Dim rangoValor As Range

    On Error GoTo ErrorConfigurarValor

    ConfigurarValorPartida = False

    Set rangoValor = ws.Cells(12, "F").Resize(500, 1)

    On Error Resume Next
    rangoValor.Validation.Delete
    On Error GoTo ErrorConfigurarValor

    With rangoValor.Validation
        .Add Type:=xlValidateCustom, _
             AlertStyle:=xlValidAlertStop, _
             Operator:=xlBetween, _
             Formula1:="=AND(ISERROR(FIND("","",F12)),ISNUMBER(VALUE(F12)),VALUE(F12)<>0)"

        .IgnoreBlank = True
        .InCellDropdown = False

        .InputTitle = "Valor partida"
        .InputMessage = "Ingrese un valor numérico diferente de cero. Use punto para decimales."

        .ErrorTitle = "Valor partida inválido"
        .ErrorMessage = "El valor partida debe ser numérico, diferente de cero y usar punto decimal."

        .ShowInput = True
        .ShowError = True
    End With

    ConfigurarValorPartida = True
    Exit Function

ErrorConfigurarValor:

    MsgBox "Error configurando la validación de Valor partida" & vbCrLf & vbCrLf & _
           "Hoja: " & ws.Name & vbCrLf & _
           "Detalle: " & Err.Description, _
           vbCritical, _
           "Error Columna F"

End Function

Sub ConfigurarValidacionesEntradaEnHoja(ByVal ws As Worksheet)

    Dim estabaProtegida As Boolean

    On Error GoTo ErrorConfigurar

    If ws.Name = "ALCON" Then Exit Sub
    If ws.Name = "MENÚ" Then Exit Sub
    If ws.Name = "DATOS" Then Exit Sub
    If ws.Name = "PLANTILLA" Then Exit Sub

    estabaProtegida = ws.ProtectContents

    If estabaProtegida Then
        ws.Unprotect Password:=CLAVE_HOJA
    End If

    ws.Range("A12:H500").Validation.Delete
    ws.Range("U12:V500").Validation.Delete

    Call ConfigurarCodigoOficina(ws)
    Call ConfigurarFechaOrigenPartida(ws)
    Call ConfigurarTipoCliente(ws)
    Call ConfigurarOficinaReclasificacion(ws)
    Call ConfigurarValorPartida(ws)
    Call ConfigurarPartidaReclasificar(ws)

    Exit Sub

SalidaSegura:

    If estabaProtegida Then
        Call ProtegerHojaTrabajo(ws)
    End If

    Exit Sub

ErrorConfigurar:

    MsgBox "No se pudo configurar las validaciones de entrada en la hoja " & ws.Name & "." & vbCrLf & vbCrLf & _
           "Detalle(s): " & Err.Description, _
           vbCritical, _
           "Error configurando validaciones"

    Resume SalidaSegura

End Sub

Sub RestaurarValidacionesBasicasEnHoja(ByVal ws As Worksheet)

    If ws.Name = "MENÚ" Then Exit Sub
    If ws.Name = "DATOS" Then Exit Sub
    If ws.Name = "ALCON" Then Exit Sub
    If ws.Name = "PLANTILLA" Then Exit Sub

    Call ConfigurarCodigoOficina(ws)
    Call ConfigurarFechaOrigenPartida(ws)
    Call ConfigurarTipoCliente(ws)
    Call ConfigurarOficinaReclasificacion(ws)
    Call ConfigurarValorPartida(ws)
    Call ConfigurarPartidaReclasificar(ws)

End Sub

Function EsPartidaReclasificarValida(ByVal valor As Variant) As Boolean

    Dim texto As String

    EsPartidaReclasificarValida = False
    If IsError(valor) Then Exit Function

    texto = LimpiarTextoValidacion(valor)

    'No se agrega obligatoriedad a V; se valida su contenido cuando existe'
    If texto = "" Or texto = "SÍ" Or texto = "NO" Then
        EsPartidaReclasificarValida = True
    End If

End Function

Function ConfigurarPartidaReclasificar(ByVal ws As Worksheet) As Boolean

    Dim rangoPartida As Range

    On Error GoTo ErrorConfigurarPartida

    ConfigurarPartidaReclasificar = False
    Set rangoPartida = ws.Cells(12, "V").Resize(500, 1)

    On Error Resume Next
    rangoPartida.Validation.Delete
    On Error GoTo ErrorConfigurarPartida

    With rangoPartida.Validation
        .Add Type:=xlValidateList, _
             AlertStyle:=xlValidAlertStop, _
             Operator:=xlBetween, _
             Formula1:="Sí,No"

        .IgnoreBlank = True
        .InCellDropdown = True
        .InputTitle = "Partida a reclasificar"
        .InputMessage = "Seleccione Sí o No"
        .ErrorTitle = "Partida a reclasificar inválida"
        .ErrorMessage = "Debe seleccionar únicamente Sí o No"
        .ShowInput = True
        .ShowError = True
    End With

    ConfigurarPartidaReclasificar = True
    Exit Function

ErrorConfigurarPartida:

    MsgBox "Error configurando Partida a reclasificar en la hoja: " & ws.Name & vbCrLf & vbCrLf & _
           "Detalle: " & Err.Description, _
           vbCritical, _
           "Error Columna V"

End Function
