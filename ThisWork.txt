'ThisWorkbook - adaptación de los dos eventos originales.
'Reemplazar el código de ThisWorkbook, no pegar en un módulo estándar.
'Requiere el Módulo 4 integrado; no ejecutar todavía en producción.
'CLAVE_HOJA debe existir una sola vez como Public Const en el Módulo 4.
'Conserva filas 12:500 y la validación antes de guardar SOLO de la hoja activa.
'Guardar desde ALCON, MENÚ, DATOS o PLANTILLA sigue omitiendo la validación.
'Pendiente en Módulo 4: corregir ValidarTodo_ReglasNegocio para conservar eventos,
'pantalla y protección, y propagar errores internos en lugar de ocultarlos.
'La Parte 1 actual fuerza EnableEvents=True: este evento lo desactiva al volver,
'pero no puede controlar lo que ocurra dentro del procedimiento llamado.
'La fórmula de H y su protección permanecen a cargo de ProtegerHojaTrabajo.
'Revisión estática únicamente; pendiente de compilación y pruebas en Excel.

Private Sub Workbook_SheetChange(ByVal Sh As Object, ByVal Target As Range)

    Dim estabaProtegida As Boolean
    Dim eventosAnteriores As Boolean
    Dim pantallaAnterior As Boolean
    Dim detalleError As String

    If Not TypeOf Sh Is Worksheet Then Exit Sub
    If Sh.Name = "ALCON" Then Exit Sub
    If Sh.Name = "MENÚ" Then Exit Sub
    If Sh.Name = "DATOS" Then Exit Sub
    If Sh.Name = "PLANTILLA" Then Exit Sub

    If Intersect(Target, Union(Sh.Range("A12:G500"), Sh.Range("U12:V500"))) Is Nothing Then Exit Sub

    eventosAnteriores = Application.EnableEvents
    pantallaAnterior = Application.ScreenUpdating
    estabaProtegida = Sh.ProtectContents

    On Error GoTo ErrorHandler
    Application.EnableEvents = False
    Application.ScreenUpdating = False

    If estabaProtegida Then
        Sh.Unprotect Password:=CLAVE_HOJA
        If Sh.ProtectContents Then Err.Raise vbObjectError + 1001, , "No se pudo desproteger la hoja."
    End If

    Call ActualizarListaRedSegunTipoCliente(Sh, Target)
    Call RestaurarValidacionesBasicasEnHoja(Sh)

SalidaSegura:

    'Mantener eventos desactivados también durante la restauración de H.
    On Error Resume Next
    Application.EnableEvents = False
    Err.Clear

    If estabaProtegida Then
        Call ProtegerHojaTrabajo(Sh)
        If Err.Number <> 0 Then
            detalleError = detalleError & vbCrLf & "Error al restaurar protección: " & Err.Description
        End If
        Err.Clear
        If Not Sh.ProtectContents Then
            detalleError = detalleError & vbCrLf & "ATENCIÓN: la hoja continúa desprotegida."
        End If
    End If

    Err.Clear
    Application.ScreenUpdating = pantallaAnterior
    If Err.Number <> 0 Then detalleError = detalleError & vbCrLf & Err.Description
    Err.Clear
    Application.EnableEvents = eventosAnteriores
    If Err.Number <> 0 Then detalleError = detalleError & vbCrLf & Err.Description
    On Error GoTo 0

    If detalleError <> "" Then
        MsgBox "No se completó la actualización de la hoja " & Sh.Name & "." & vbCrLf & _
            detalleError, vbCritical, "Error actualizando hoja"
    End If
    Exit Sub

ErrorHandler:

    detalleError = "Error " & Err.Number & ": " & Err.Description
    Resume SalidaSegura

End Sub

Private Sub Workbook_BeforeSave(ByVal SaveAsUI As Boolean, Cancel As Boolean)

    Dim ws As Worksheet
    Dim paso As String
    Dim estabaProtegida As Boolean
    Dim eventosAnteriores As Boolean
    Dim pantallaAnterior As Boolean
    Dim detalleError As String

    'El validador original utiliza ActiveSheet; no validar otro libro por error.
    If Not TypeOf ActiveSheet Is Worksheet Then
        Cancel = True
        MsgBox "Active una hoja de cálculo antes de guardar.", vbExclamation, "Guardado bloqueado"
        Exit Sub
    End If

    Set ws = ActiveSheet
    If Not (ws.Parent Is Me) Then
        Cancel = True
        MsgBox "Active una hoja de este libro antes de guardar.", vbExclamation, "Guardado bloqueado"
        Exit Sub
    End If

    If ws.Name = "ALCON" Then Exit Sub
    If ws.Name = "MENÚ" Then Exit Sub
    If ws.Name = "DATOS" Then Exit Sub
    If ws.Name = "PLANTILLA" Then Exit Sub

    eventosAnteriores = Application.EnableEvents
    pantallaAnterior = Application.ScreenUpdating
    estabaProtegida = ws.ProtectContents

    On Error GoTo ErrorHandler
    paso = "Preparar la hoja activa"
    Application.EnableEvents = False
    Application.ScreenUpdating = False

    If estabaProtegida Then
        ws.Unprotect Password:=CLAVE_HOJA
        If ws.ProtectContents Then Err.Raise vbObjectError + 1001, , "No se pudo desproteger la hoja."
    End If

    paso = "Ejecutar reglas de negocio"
    Call ValidarTodo_ReglasNegocio

    'Necesario mientras el validador original reactive eventos al finalizar.
    Application.EnableEvents = False
    Application.ScreenUpdating = False

    paso = "Revisar errores en rojo"
    If ExistenErroresMarcadosEnRojo(ws) Then
        Cancel = True
        MsgBox "No es posible guardar el archivo porque existen celdas marcadas en rojo." & vbCrLf & _
            "Corrige los errores y vuelve a ejecutar la validación antes de guardar.", _
            vbCritical, "Guardado bloqueado"
    End If

SalidaSegura:

    On Error Resume Next
    Application.EnableEvents = False
    Err.Clear

    If estabaProtegida Then
        Call ProtegerHojaTrabajo(ws)
        If Err.Number <> 0 Then
            Cancel = True
            detalleError = detalleError & vbCrLf & "Error al restaurar protección: " & Err.Description
        End If
        Err.Clear
        If Not ws.ProtectContents Then
            Cancel = True
            detalleError = detalleError & vbCrLf & "ATENCIÓN: la hoja continúa desprotegida."
        End If
    End If

    Err.Clear
    Application.ScreenUpdating = pantallaAnterior
    If Err.Number <> 0 Then detalleError = detalleError & vbCrLf & Err.Description
    Err.Clear
    Application.EnableEvents = eventosAnteriores
    If Err.Number <> 0 Then detalleError = detalleError & vbCrLf & Err.Description
    On Error GoTo 0

    If detalleError <> "" Then
        Cancel = True
        MsgBox "No se completó la validación automática antes de guardar." & vbCrLf & _
            detalleError, vbCritical, "Guardado bloqueado"
    End If
    Exit Sub

ErrorHandler:

    Cancel = True
    detalleError = "Paso: " & paso & vbCrLf & "Error " & Err.Number & ": " & Err.Description
    Resume SalidaSegura

End Sub
